● REST API · /api/v1 · self-hosted server

Make reports from your own systems.

The self-hosted ReportWright server has a REST API. Send a report definition or a template name and a JSON payload, get back a PDF, an Excel file, Word, PowerPoint, HTML or CSV. Run a batch of thousands of documents in the background, or put one on a cron schedule. Every example on this page uses the request and response shapes in the API reference.

API reference (docs/API.md) Self-host guide (deploy/compose)

Keys: set PW_API_KEYS on your server and send one with each request as Authorization: Bearer <key> or x-api-key. The examples read the key from PW_API_KEY in your shell. Base URL: replace https://reports.example.com with your server.

Render one report to PDF

POST /api/v1/render. The answer is the file itself. The page count comes back in the x-reportwright-pages header. Errors are JSON { error }; a busy server answers 429 or 503 with Retry-After.

curl -X POST https://reports.example.com/api/v1/render \
  -H "Authorization: Bearer $PW_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "template": "loan-statement",
    "format": "pdf",
    "parameters": { "accountId": "LN-2025-001982" },
    "fileName": "statement-LN-2025-001982"
  }' \
  --output statement.pdf
import fs from 'node:fs';

const res = await fetch('https://reports.example.com/api/v1/render', {
  method: 'POST',
  headers: { authorization: `Bearer ${process.env.PW_API_KEY}`, 'content-type': 'application/json' },
  body: JSON.stringify({
    template: 'loan-statement',
    format: 'pdf',
    parameters: { accountId: 'LN-2025-001982' },
    fileName: 'statement-LN-2025-001982',
  }),
});
if (!res.ok) throw new Error(`HTTP ${res.status}: ${await res.text()}`);
fs.writeFileSync('statement.pdf', Buffer.from(await res.arrayBuffer()));
import os
import requests

r = requests.post(
    "https://reports.example.com/api/v1/render",
    headers={"Authorization": f"Bearer {os.environ['PW_API_KEY']}"},
    json={
        "template": "loan-statement",
        "format": "pdf",
        "parameters": {"accountId": "LN-2025-001982"},
        "fileName": "statement-LN-2025-001982",
    },
    timeout=120,
)
r.raise_for_status()
with open("statement.pdf", "wb") as f:
    f.write(r.content)

Start a batch, poll it, download the zip

POST /api/v1/batches answers 202 with the job, which runs in the background. Poll GET /api/v1/batches/<id> until the status is no longer queued or running, then fetch /download. One failed item does not stop the batch; its error is in errors[].

# 1. start the batch (202 with the job)
ID=$(curl -s -X POST https://reports.example.com/api/v1/batches \
  -H "Authorization: Bearer $PW_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "template": "loan-statement",
    "format": "pdf",
    "fileName": "statement-{Parameters.accountId}",
    "items": [
      { "parameters": { "accountId": "LN-2026-004417" } },
      { "parameters": { "accountId": "LN-2025-001982" } }
    ],
    "output": { "type": "zip" }
  }' | jq -r '.id')

# 2. poll until it is no longer queued or running
while true; do
  STATUS=$(curl -s https://reports.example.com/api/v1/batches/$ID \
    -H "Authorization: Bearer $PW_API_KEY" | jq -r '.status')
  case "$STATUS" in queued|running) sleep 2 ;; *) break ;; esac
done
echo "status: $STATUS"

# 3. download the zip (completed or completed_with_errors)
curl -s https://reports.example.com/api/v1/batches/$ID/download \
  -H "Authorization: Bearer $PW_API_KEY" --output statements.zip
import fs from 'node:fs';

const BASE = 'https://reports.example.com';
const headers = { authorization: `Bearer ${process.env.PW_API_KEY}` };

// 1. start the batch (202 with the job)
const start = await fetch(`${BASE}/api/v1/batches`, {
  method: 'POST',
  headers: { ...headers, 'content-type': 'application/json' },
  body: JSON.stringify({
    template: 'loan-statement',
    format: 'pdf',
    fileName: 'statement-{Parameters.accountId}',
    items: [
      { parameters: { accountId: 'LN-2026-004417' } },
      { parameters: { accountId: 'LN-2025-001982' } },
    ],
    output: { type: 'zip' },
  }),
});
if (start.status !== 202) throw new Error(`HTTP ${start.status}: ${await start.text()}`);
const { id } = await start.json();

// 2. poll until it is no longer queued or running
let job;
do {
  await new Promise((r) => setTimeout(r, 2000));
  job = await (await fetch(`${BASE}/api/v1/batches/${id}`, { headers })).json();
} while (job.status === 'queued' || job.status === 'running');
console.log(job.status, 'done:', job.done, 'failed:', job.failed, job.errors);

// 3. download the zip
if (job.status === 'completed' || job.status === 'completed_with_errors') {
  const zip = await fetch(`${BASE}/api/v1/batches/${id}/download`, { headers });
  fs.writeFileSync('statements.zip', Buffer.from(await zip.arrayBuffer()));
}
import os
import time
import requests

BASE = "https://reports.example.com"
headers = {"Authorization": f"Bearer {os.environ['PW_API_KEY']}"}

# 1. start the batch (202 with the job)
start = requests.post(f"{BASE}/api/v1/batches", headers=headers, timeout=60, json={
    "template": "loan-statement",
    "format": "pdf",
    "fileName": "statement-{Parameters.accountId}",
    "items": [
        {"parameters": {"accountId": "LN-2026-004417"}},
        {"parameters": {"accountId": "LN-2025-001982"}},
    ],
    "output": {"type": "zip"},
})
start.raise_for_status()
job_id = start.json()["id"]

# 2. poll until it is no longer queued or running
while True:
    job = requests.get(f"{BASE}/api/v1/batches/{job_id}", headers=headers, timeout=30).json()
    if job["status"] not in ("queued", "running"):
        break
    time.sleep(2)
print(job["status"], "done:", job["done"], "failed:", job["failed"], job.get("errors"))

# 3. download the zip
if job["status"] in ("completed", "completed_with_errors"):
    z = requests.get(f"{BASE}/api/v1/batches/{job_id}/download", headers=headers, timeout=300)
    z.raise_for_status()
    with open("statements.zip", "wb") as f:
        f.write(z.content)

Schedule a report

POST /api/v1/schedules answers 201. A schedule runs a batch on a cron expression in a time zone. Cron supports *, lists, ranges and */n; use numbers for days and months, not names. Also: GET /api/v1/schedules (each with nextRunAt), and POST /api/v1/schedules/<id>/run to run one now.

curl -X POST https://reports.example.com/api/v1/schedules \
  -H "Authorization: Bearer $PW_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Monthly statements",
    "template": "loan-statement",
    "cron": "0 7 1 * *",
    "timezone": "Asia/Kolkata",
    "itemsFrom": { "url": "/api/sample/loans", "path": "$[*]", "map": { "accountId": "$.id" } },
    "output": { "type": "folder", "folder": "monthly" },
    "fileName": "{Parameters.accountId}",
    "webhook": "https://your-app/hooks/reportwright",
    "enabled": true
  }'
const res = await fetch('https://reports.example.com/api/v1/schedules', {

Health

GET /api/v1/health with a service key. It reports the render pool (size, alive, ready, busy), queue depth by class, the estimated wait in ms, and totals since start (done, failed, rejected, timedOut, crashed, recycled, dropped).

curl https://reports.example.com/api/v1/health \
  -H "Authorization: Bearer $PW_API_KEY"
const res = await fetch('https://reports.example.com/api/v1/health', {
  headers: { authorization: `Bearer ${process.env.PW_API_KEY}` },
});
console.log(await res.json());
import os
import requests

r = requests.get(
    "https://reports.example.com/api/v1/health",
    headers={"Authorization": f"Bearer {os.environ['PW_API_KEY']}"},
    timeout=10,
)
print(r.json())
